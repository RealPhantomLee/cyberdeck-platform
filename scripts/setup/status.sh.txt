#!/usr/bin/env bash
set -euo pipefail

# Brick 2 - Cyberdeck: Quick health/status report (safe)
# Usage:
#   ./status.sh

hr() { printf '%*s\n' "${COLUMNS:-80}" '' | tr ' ' '-'; }

echo "Cyberdeck Status — $(date -Is)"
hr

# OS + host
echo "Host:      $(hostname)"
echo "User:      $(whoami)"
echo "OS:        $(. /etc/os-release 2>/dev/null && echo "${PRETTY_NAME}" || uname -a)"
echo "Kernel:    $(uname -r)"
echo "Uptime:    $(uptime -p 2>/dev/null || uptime)"
hr

# CPU + memory
if command -v lscpu >/dev/null 2>&1; then
  echo "CPU:       $(lscpu | awk -F: '/Model name/ {gsub(/^[ \t]+/, "", $2); print $2; exit}')"
fi

# Temperature (Pi or generic)
TEMP=""
if command -v vcgencmd >/dev/null 2>&1; then
  TEMP="$(vcgencmd measure_temp | sed 's/temp=//')"
elif [[ -r /sys/class/thermal/thermal_zone0/temp ]]; then
  t="$(cat /sys/class/thermal/thermal_zone0/temp)"
  TEMP="$(awk -v t="$t" 'BEGIN {printf "%.1f°C\n", t/1000}')"
fi
echo "Temp:      ${TEMP:-N/A}"

# Throttle flags (Raspberry Pi)
if command -v vcgencmd >/dev/null 2>&1; then
  echo "Pi Flags:  $(vcgencmd get_throttled 2>/dev/null || echo N/A)"
else
  echo "Pi Flags:  N/A"
fi

echo "Memory:    $(free -h | awk '/Mem:/ {print $3 " used / " $2 " total"}')"
hr

# Disk
echo "Disk usage:"
df -h -x tmpfs -x devtmpfs | awk 'NR==1{print} NR>1{print}'
hr

# Network summary (no scanning)
echo "Network interfaces:"
if command -v ip >/dev/null 2>&1; then
  ip -brief addr
else
  ifconfig 2>/dev/null || true
fi
hr

# Services (minimal)
for svc in ssh sshd NetworkManager; do
  if systemctl list-unit-files | grep -q "^${svc}\.service"; then
    state="$(systemctl is-active "${svc}" 2>/dev/null || echo "unknown")"
    echo "Service ${svc}: ${state}"
  fi
done
hr

echo "Done."
